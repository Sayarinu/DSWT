cmake_minimum_required(VERSION 3.22)
project(platinumwt_emulator_persistence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(platinumwt_durable_save STATIC durable_save.cpp)
target_include_directories(platinumwt_durable_save PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(platinumwt_durable_save PRIVATE OpenSSL::Crypto)
target_compile_options(platinumwt_durable_save PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(platinumwt-save-store save_store.cpp)
target_link_libraries(platinumwt-save-store PRIVATE platinumwt_durable_save)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(persistence-probe tests/probe.cpp)
    target_link_libraries(persistence-probe PRIVATE platinumwt_durable_save)
    add_library(persistence-faults SHARED tests/faults.cpp)
    target_link_libraries(persistence-faults PRIVATE dl)
    add_test(NAME persistence COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_persistence.py"
        $<TARGET_FILE:platinumwt-save-store> $<TARGET_FILE:persistence-probe> $<TARGET_FILE:persistence-faults>)
    set_tests_properties(persistence PROPERTIES TIMEOUT 60)
endif()
