cmake_minimum_required(VERSION 3.22)
project(dswt_emulator_persistence LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(dswt_durable_save STATIC durable_save.cpp)
target_include_directories(dswt_durable_save PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(dswt_durable_save PRIVATE OpenSSL::Crypto)
target_compile_options(dswt_durable_save PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(dswt_journal STATIC ../client/src/wonder_trade/journal.c)
target_include_directories(dswt_journal PUBLIC ../client/include)
target_compile_options(dswt_journal PRIVATE -Wall -Wextra -Werror)
add_library(dswt_journal_seed STATIC journal_seed.cpp)
target_link_libraries(dswt_journal_seed PUBLIC dswt_durable_save dswt_journal OpenSSL::Crypto)
target_compile_options(dswt_journal_seed PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(dswt-save-store save_store.cpp)
target_link_libraries(dswt-save-store PRIVATE dswt_durable_save)
target_link_libraries(dswt-save-store PRIVATE dswt_journal_seed)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(persistence-probe tests/probe.cpp)
    target_link_libraries(persistence-probe PRIVATE dswt_durable_save)
    add_library(persistence-faults SHARED tests/faults.cpp)
    add_executable(journal-fixture tests/journal_fixture.cpp ../client/src/wonder_trade/wire.c)
    target_link_libraries(journal-fixture PRIVATE dswt_journal OpenSSL::Crypto)
    target_link_libraries(persistence-faults PRIVATE dl)
    add_test(NAME persistence COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_persistence.py"
        $<TARGET_FILE:dswt-save-store> $<TARGET_FILE:persistence-probe> $<TARGET_FILE:persistence-faults>)
    set_tests_properties(persistence PROPERTIES TIMEOUT 60)
    add_test(NAME journal_seed COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_journal_seed.py"
        $<TARGET_FILE:dswt-save-store> $<TARGET_FILE:journal-fixture> $<TARGET_FILE:persistence-faults>)
    set_tests_properties(journal_seed PROPERTIES TIMEOUT 180)
endif()
