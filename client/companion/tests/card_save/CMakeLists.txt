foreach(mode readonly writable)
    foreach(suite card_save spi)
        set(target ${suite}_${mode}_test)
        if(suite STREQUAL "spi")
            add_executable(${target} spi_test.c "${COMPANION_SRC}/platform/nds/card_save_spi.c")
            target_include_directories(${target} PRIVATE registers)
        else()
            add_executable(${target} card_save_test.c sim.c "${COMPANION_SRC}/platform/nds/card_save.c")
        endif()
        target_include_directories(${target} PRIVATE "${COMPANION_SRC}")
        if(mode STREQUAL "writable")
            target_compile_definitions(${target} PRIVATE CP_CARD_SAVE_WRITES=1)
        endif()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic -fsanitize=address,undefined)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
endforeach()
