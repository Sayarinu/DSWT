cmake_minimum_required(VERSION 3.22)
project(wonder_trade_tls_tests C)

set(CMAKE_C_STANDARD 99)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../include/wonder_trade/tls_config.h" CACHE FILEPATH "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../include/wonder_trade/crypto_config.h" CACHE FILEPATH "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)

add_library(wt_transport STATIC ../src/wonder_trade/tls.c ../src/wonder_trade/http.c ../src/wonder_trade/seed.c ../src/wonder_trade/journal.c)
target_include_directories(wt_transport PUBLIC ../include)
target_link_libraries(wt_transport PUBLIC mbedtls)
target_compile_options(wt_transport PRIVATE -Wall -Wextra -Werror)
add_executable(tls_probe tls_probe.c)
target_link_libraries(tls_probe PRIVATE wt_transport)
target_compile_options(tls_probe PRIVATE -Wall -Wextra -Werror)
add_executable(http_test http_test.c ../src/wonder_trade/http.c)
target_include_directories(http_test PRIVATE ../include)
target_compile_options(http_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(http_test PRIVATE -fsanitize=address,undefined)
enable_testing()
add_test(NAME http_test COMMAND http_test)
add_executable(seed_test seed_test.c)
target_link_libraries(seed_test PRIVATE wt_transport)
target_compile_options(seed_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME seed_test COMMAND seed_test)
add_executable(journal_test journal_test.c ../src/wonder_trade/journal.c)
target_link_libraries(journal_test PRIVATE wt_transport)
target_compile_options(journal_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(journal_test PRIVATE -fsanitize=address,undefined)
add_test(NAME journal_test COMMAND journal_test)
set_tests_properties(journal_test PROPERTIES TIMEOUT 180)
