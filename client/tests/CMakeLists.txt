cmake_minimum_required(VERSION 3.22)
project(wonder_trade_tls_tests C)

# These test executables use assert() for their checks, including release builds.
add_compile_options(-UNDEBUG)
set(CMAKE_C_STANDARD 99)
if(APPLE)
    add_compile_definitions(_DARWIN_C_SOURCE)
endif()
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../include/wonder_trade/tls_config.h" CACHE FILEPATH "" FORCE)

set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
# With certificate time checks disabled for the DS, pinned upstream passes an
# unused time value. New Apple Clang diagnoses it; keep the warning visible.
if(CMAKE_C_COMPILER_ID STREQUAL "AppleClang")
    include(CheckCCompilerFlag)
    check_c_compiler_flag("-Wno-error=uninitialized-const-pointer" HAS_CONST_POINTER_WARNING)
    if(HAS_CONST_POINTER_WARNING)
        target_compile_options(mbedx509 PRIVATE -Wno-error=uninitialized-const-pointer)
    endif()
endif()
